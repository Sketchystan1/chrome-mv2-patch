cmake_minimum_required(VERSION 3.24)
project(mv2_autoscan VERSION 1.16.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The compact scanner requires MSVC on Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "" FORCE)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
string(REPLACE "/GR" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
set(CMAKE_CXX_FLAGS_RELEASE "/O1 /Os /DNDEBUG")
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/GL> $<$<COMPILE_LANGUAGE:CXX>:/GS->
  $<$<COMPILE_LANGUAGE:CXX>:/GR-> $<$<COMPILE_LANGUAGE:CXX>:/EHs-c->
  $<$<COMPILE_LANGUAGE:CXX>:/Gw> $<$<COMPILE_LANGUAGE:CXX>:/Gy>)
add_link_options(/LTCG /OPT:REF /OPT:ICF /INCREMENTAL:NO)
add_compile_definitions(UNICODE _UNICODE _WIN32_WINNT=0x0A00)

string(TOLOWER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" arch)
if(arch STREQUAL "arm64")
  set(detours_arch _ARM64_)
  set(disol disolarm64.cpp)
  set(assembler armasm64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(arch x64)
  set(detours_arch _AMD64_)
  set(disol disolx64.cpp)
  set(assembler ml64)
else()
  set(arch x86)
  set(detours_arch _X86_)
  set(disol disolx86.cpp)
  set(assembler ml)
endif()
include(FetchContent)
FetchContent_Declare(detours_src
  URL https://codeload.github.com/microsoft/Detours/tar.gz/e4bfd6b03e50de46b47abfbd1e46b384f0c5f833
  URL_HASH SHA256=d24f5dd1f977c33301ebd9f3be064e938a7908954df15b31235c9ca6691cb356)
FetchContent_MakeAvailable(detours_src)
configure_file("${detours_src_SOURCE_DIR}/LICENSE.md" "${CMAKE_CURRENT_BINARY_DIR}/Detours-LICENSE.md" COPYONLY)
# Compile the unchanged import enumerator and its helper from the pinned source.
# Whole modules.cpp also references Detours' allocating inline-hook implementation.
file(READ "${detours_src_SOURCE_DIR}/src/modules.cpp" detours_modules)
set(detours_imports "// Copyright (c) Microsoft Corporation. Licensed under the MIT license.\n#include <windows.h>\n#include \"detours.h\"\n")
foreach(part containing imports)
  if(part STREQUAL "containing")
    set(first "HMODULE WINAPI DetourGetContainingModule(")
    set(last "BOOL WINAPI DetourEnumerateExports(")
  else()
    set(first "BOOL WINAPI DetourEnumerateImportsEx(")
    set(last "// Context for DetourEnumerateImportsThunk")
  endif()
  string(FIND "${detours_modules}" "${first}" begin)
  string(FIND "${detours_modules}" "${last}" end)
  if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Pinned Detours import implementation was not found")
  endif()
  math(EXPR length "${end} - ${begin}")
  string(SUBSTRING "${detours_modules}" ${begin} ${length} body)
  string(APPEND detours_imports "${body}")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/detours-imports.cpp" CONTENT "${detours_imports}" @ONLY)
add_library(detours STATIC "${CMAKE_CURRENT_BINARY_DIR}/detours-imports.cpp")
target_include_directories(detours PUBLIC ${detours_src_SOURCE_DIR}/src)
target_compile_definitions(detours PRIVATE ${detours_arch})
target_compile_options(detours PRIVATE /FIintrin.h)

add_library(autoscan_core STATIC autoscan.cpp)
target_link_libraries(autoscan_core PUBLIC bcrypt)
add_library(autoscan_apply STATIC autoscan_apply.cpp autoscan_cache.cpp)
target_link_libraries(autoscan_apply PUBLIC autoscan_core detours crypt32)
add_executable(mv2-scan autoscan_cli.cpp)
target_link_libraries(mv2-scan PRIVATE autoscan_core)

# Tail jumps preserve both documented and undocumented export ABIs without writable code.
set(exports GetFileVersionInfoA GetFileVersionInfoByHandle GetFileVersionInfoExA GetFileVersionInfoExW
  GetFileVersionInfoSizeA GetFileVersionInfoSizeExA GetFileVersionInfoSizeExW GetFileVersionInfoSizeW
  GetFileVersionInfoW VerFindFileA VerFindFileW VerInstallFileA VerInstallFileW
  VerLanguageNameA VerLanguageNameW VerQueryValueA VerQueryValueW)
set(definition "LIBRARY version\nEXPORTS\n")
if(arch STREQUAL "arm64")
  set(assembly " AREA |.text|, CODE, READONLY, ALIGN=2\n IMPORT g_versionExports\n")
elseif(arch STREQUAL "x64")
  set(assembly "EXTERN g_versionExports:QWORD\n.code\n")
else()
  set(assembly ".386\n.model flat\nEXTERN _g_versionExports:DWORD\n.code\n")
endif()
set(ordinal 1)
foreach(name IN LISTS exports)
  math(EXPR offset "(${ordinal} - 1) * ${CMAKE_SIZEOF_VOID_P}")
  set(symbol "mv2_export_${ordinal}")
  string(APPEND definition " ${name}=${symbol} @${ordinal}\n")
  if(arch STREQUAL "x86")
    string(PREPEND symbol "_")
  endif()
  if(arch STREQUAL "arm64")
    string(APPEND assembly " EXPORT ${symbol}\n${symbol} PROC\n adrp x16,g_versionExports\n add x16,x16,g_versionExports\n ldr x16,[x16,#${offset}]\n br x16\n ENDP\n")
  elseif(arch STREQUAL "x64")
    string(APPEND assembly "PUBLIC ${symbol}\n${symbol} PROC\n jmp QWORD PTR [g_versionExports+${offset}]\n${symbol} ENDP\n")
  else()
    string(APPEND assembly "PUBLIC ${symbol}\n${symbol} PROC\n jmp DWORD PTR [_g_versionExports+${offset}]\n${symbol} ENDP\n")
  endif()
  math(EXPR ordinal "${ordinal} + 1")
endforeach()
if(arch STREQUAL "x86")
  string(APPEND definition " VersionMV2ScanW=_VersionMV2ScanW@16 @18\n")
else()
  string(APPEND definition " VersionMV2ScanW @18\n")
endif()
string(APPEND assembly " END\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/version.def" CONTENT "${definition}" @ONLY)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/version-exports.asm" CONTENT "${assembly}" @ONLY)
get_filename_component(compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(MV2_ASSEMBLER NAMES ${assembler} HINTS "${compiler_dir}" REQUIRED)
set(export_object "${CMAKE_CURRENT_BINARY_DIR}/version-exports.obj")
if(arch STREQUAL "arm64")
  add_custom_command(OUTPUT "${export_object}" COMMAND "${MV2_ASSEMBLER}" -nologo -o "${export_object}" "${CMAKE_CURRENT_BINARY_DIR}/version-exports.asm"
    DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/version-exports.asm" VERBATIM)
else()
  add_custom_command(OUTPUT "${export_object}" COMMAND "${MV2_ASSEMBLER}" /nologo /c "/Fo${export_object}" "${CMAKE_CURRENT_BINARY_DIR}/version-exports.asm"
    DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/version-exports.asm" VERBATIM)
endif()
add_library(version SHARED proxy.cpp autoscan_runtime.cpp version.rc "${export_object}" "${CMAKE_CURRENT_BINARY_DIR}/version.def")
set_target_properties(version PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build/${arch}$<0:>"
  ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
  PDB_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(version PRIVATE MV2_VER_STR="${PROJECT_VERSION}.0"
  MV2_VERC=${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${PROJECT_VERSION_PATCH},0 MV2_PROD="${PROJECT_VERSION}")
target_link_libraries(version PRIVATE autoscan_apply delayimp)
target_link_options(version PRIVATE /DYNAMICBASE /NXCOMPAT /MANIFEST:NO /DELAYLOAD:crypt32.dll)
foreach(target version mv2-scan)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND}
    "-DBINARY=$<TARGET_FILE:${target}>" -P "${CMAKE_CURRENT_SOURCE_DIR}/check-size.cmake" VERBATIM)
endforeach()
